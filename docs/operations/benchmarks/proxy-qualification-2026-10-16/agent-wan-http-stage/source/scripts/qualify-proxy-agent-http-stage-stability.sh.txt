stop_owned() {
  local attempt
  [[ -n $owned_pid ]] || return 0
  kill -TERM -- "-$owned_pid" 2>/dev/null || true
  for ((attempt=0; attempt<100; attempt++)); do
    kill -0 -- "-$owned_pid" 2>/dev/null || break
    sleep .1
  done
  kill -KILL -- "-$owned_pid" 2>/dev/null || true
  wait "$owned_pid" 2>/dev/null || true
}
write_exit() {
  local status=$1 timed_out=$2 interrupted=$3
  python3 - "$output_dir/exit.json" "$status" "$timed_out" "$interrupted" <<'PY'

path,status,timed_out,interrupted=sys.argv[1:]

PY
}

  local status=$?
  ((finished)) || { stop_owned; write_exit "$status" 0 0; }
