func terminalTestHeader(kind byte, flags uint16, stream, value uint32) []byte {
	h := make([]byte, 12)
	h[1] = kind
	binary.BigEndian.PutUint16(h[2:4], flags)
	binary.BigEndian.PutUint32(h[4:8], stream)
	binary.BigEndian.PutUint32(h[8:12], value)
	return h
}

func TestQualificationTerminalControlParser(t *testing.T) {
	positives := map[string][]byte{
		"goaway_normal": terminalTestHeader(3, 0, 0, 0),
		"ping_syn":      terminalTestHeader(2, 1, 0, 7), "ping_ack": terminalTestHeader(2, 2, 0, 7),
		"stream_credit": terminalTestHeader(1, 0, 1, 32768), "stream_syn": terminalTestHeader(1, 1, 1, 0),
		"stream_ack": terminalTestHeader(1, 2, 1, 0), "stream_fin": terminalTestHeader(1, 4, 1, 0),
		"stream_syn_fin": terminalTestHeader(1, 5, 1, 0), "stream_ack_fin": terminalTestHeader(1, 6, 1, 0),
		"stream_rst": terminalTestHeader(1, 8, 1, 0), "global_refund": terminalTestHeader(1, 0, 0, 4096),
		"allocation_refund": terminalTestHeader(4, 0, 1, 4096), "stream_growth": terminalTestHeader(5, 0, 1, 4096),
		"global_growth": terminalTestHeader(5, 0, 0, 4096),
	}
	batch := []byte{}
	for i := uint32(1); i <= 16; i++ {
		batch = append(batch, terminalTestHeader(4, 0, i, 4096)...)
	}
	batch = append(batch, terminalTestHeader(1, 0, 0, 65536)...)
	positives["bounded_refund_batch"] = batch
	for name, p := range positives {
		t.Run(name, func(t *testing.T) {
			headers, ok := terminalParseControls(p)
			if !ok || len(headers)*12 != len(p) {
				t.Fatal("source-emitted control rejected")
			}
			for i, h := range headers {
				want := p[i*12 : (i+1)*12]
				if h.Type != want[1] || h.Flags != binary.BigEndian.Uint16(want[2:4]) || h.Stream != binary.BigEndian.Uint32(want[4:8]) || h.Value != binary.BigEndian.Uint32(want[8:12]) {
					t.Fatal("exact frame metadata lost")
				}
			}
		})
	}
	badVersion := terminalTestHeader(3, 0, 0, 0)
	badVersion[0] = 1
	negatives := map[string][]byte{
		"empty": nil, "truncated": {0, 3}, "unknown_type": terminalTestHeader(99, 0, 0, 0), "version": badVersion,
		"empty_data": terminalTestHeader(0, 4, 1, 0), "data": append(terminalTestHeader(0, 0, 1, 1), 1),
		"mixed_data_control": append(terminalTestHeader(3, 0, 0, 0), terminalTestHeader(0, 0, 1, 0)...),
		"goaway_error":       terminalTestHeader(3, 0, 0, 1), "ping_stream": terminalTestHeader(2, 1, 1, 7),
		"ping_flags": terminalTestHeader(2, 3, 0, 7), "invalid_window_flags": terminalTestHeader(1, 3, 1, 0),
		"global_stream_flags": terminalTestHeader(1, 1, 0, 4096), "global_unaligned": terminalTestHeader(1, 0, 0, 1),
		"rst_credit": terminalTestHeader(1, 8, 1, 4096), "allocation_zero_stream": terminalTestHeader(4, 0, 0, 4096),
		"allocation_unaligned": terminalTestHeader(4, 0, 1, 1), "growth_zero": terminalTestHeader(5, 0, 1, 0),
		"excess_batch": append(append([]byte{}, batch...), terminalTestHeader(3, 0, 0, 0)...),
	}
	for name, p := range negatives {
		t.Run(name, func(t *testing.T) {
			if _, ok := terminalParseControls(p); ok {
				t.Fatal("unknown/malformed/DATA control accepted")
			}
		})
	}
}

func terminalTestRecorder() *qualificationRawWriteTerminalRecorder {
	r := &qualificationRawWriteTerminalRecorder{path: "unused", expected: 5, active: make(map[int]*yamux.Session), records: make(map[int]qualificationRawWriteTerminalRecord)}
	r.terminal.owners = make(map[int]*qualificationRawWriteOwner)
	r.terminal.identities = make(map[int]string)
	for lane := 0; lane < 5; lane++ {
		session := new(yamux.Session)
		r.active[lane] = session
		r.terminal.identities[lane] = fmt.Sprintf("%p", session)
		r.terminal.owners[lane] = &qualificationRawWriteOwner{lane: lane, generation: uint64(lane + 1), rawBound: true, bodyBound: true}
	}
	return r
}

func TestQualificationTerminalIntentCoverage(t *testing.T) {
	t.Run("five_live_owners", func(t *testing.T) {
		r := terminalTestRecorder()
		if err := r.markTerminalIntent(); err != nil {
			t.Fatal(err)
		}
		if !r.terminal.intentComplete {
			t.Fatal("coverage incomplete")
		}
		for _, owner := range r.terminal.owners {
			if owner.terminal.Stop == 0 || owner.terminal.StopGeneration != owner.generation {
				t.Fatal("owner intent missing")
			}
		}
		if r.markTerminalIntent() == nil {
			t.Fatal("duplicate intent accepted")
		}
	})
	for name, mutate := range map[string]func(*qualificationRawWriteTerminalRecorder){
		"missing_owner":        func(r *qualificationRawWriteTerminalRecorder) { delete(r.terminal.owners, 4) },
		"missing_live_session": func(r *qualificationRawWriteTerminalRecorder) { delete(r.active, 4) },
		"changed_session":      func(r *qualificationRawWriteTerminalRecorder) { r.active[4] = new(yamux.Session) },
		"closed_owner":         func(r *qualificationRawWriteTerminalRecorder) { r.terminal.owners[4].closeStarted = true },
		"active_data":          func(r *qualificationRawWriteTerminalRecorder) { r.terminal.owners[4].activeEnvelope = true },
		"unbound_owner":        func(r *qualificationRawWriteTerminalRecorder) { r.terminal.owners[4].rawBound = false },
		"unconsumed_data":      func(r *qualificationRawWriteTerminalRecorder) { r.terminal.owners[4].interval.envelopeCount = 1 },
	} {
		t.Run(name, func(t *testing.T) {
			r := terminalTestRecorder()
			mutate(r)
			if r.markTerminalIntent() == nil || r.terminal.intentComplete {
				t.Fatal("partial/invalid intent accepted")
			}
		})
	}
}

func TestQualificationTerminalReasonsAndOrdering(t *testing.T) {
	owner := &qualificationRawWriteOwner{generation: 1, activeRawWrites: 1}
	headers, _ := terminalParseControls(terminalTestHeader(3, 0, 0, 0))
	owner.terminal.ActivePlain = true
	owner.terminal.plain = terminalPlain{Begin: owner.terminal.next(), Control: true, Attempt: 12, Headers: headers}
	token := owner.terminalBeginRawLocked()
	owner.terminal.Stop = owner.terminal.next()
	owner.terminal.StopGeneration = 1
	owner.terminalPhysicalCloseBeginLocked()
	owner.terminalPhysicalCloseEnd()
	owner.terminalLegacyCloseLocked()
	if !owner.terminal.LegacyCloseOverlap || owner.terminal.Problems != 0 {
		t.Fatal("exact control close reason lost")
	}
	owner.terminalRawReturned(token, 34, 0, net.ErrClosed)
	owner.mu.Lock()
	owner.terminalFinishRawLocked(token)
	owner.mu.Unlock()
	v := owner.terminal.Records[0]
	if v.Return >= v.After || !v.Error || !v.NetClosed || v.Written != 0 || owner.terminal.RawErrors != 1 {
		t.Fatal("real return metadata or preserved error accounting lost")
	}
	owner.terminalBeginRawLocked()
	if owner.terminal.Problems&terminalPostCloseWrite == 0 {
		t.Fatal("new post-close write not independently rejected")
	}
	owner.terminal.reject(terminalOtherLegacyAmbiguity)
	if owner.terminal.Problems&terminalOtherLegacyAmbiguity == 0 {
		t.Fatal("unrelated legacy ambiguity suppressed")
	}
}
