import csv,json
from pathlib import Path
root=Path(__file__).resolve().parent
# Whitelisted post-run metadata preserves original exact tuple-to-lane mapping.
lanes=json.loads((root/'connection-inputs.json').read_text())
lookup={(r['mode'],r['remote_addr']):r['lane'] for r in lanes}
phases=json.loads((root/'tcp-phases.json').read_text())
for p in phases:
 remote=p['local'] if p['sender']=='agent' else p['peer']
 p['lane']=lookup[(p['mode'],remote)]
(root/'tcp-phases-with-lanes.json').write_text(json.dumps(phases,indent=2)+'\n')
(root/'physical-lanes.json').write_text(json.dumps(lanes,indent=2)+'\n')
with (root/'tcp-active-sender.csv').open() as f: rows=list(csv.DictReader(f))
for r in rows:
 remote=r['local'] if r['side']=='agent' else r['peer']
 r['lane']=lookup[(r['mode'],remote)]
with (root/'tcp-active-sender-with-lanes.csv').open('w') as f:
 writer=csv.DictWriter(f,fieldnames=list(rows[0])); writer.writeheader(); writer.writerows(rows)
print('Mapped',len(lanes),'physical connections and',len(phases),'transfer phases to exact logged lane IDs')
