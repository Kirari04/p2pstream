#!/usr/bin/env python3
"""Low-cost pure contracts for the diagnostic native-Go profile derivative."""
import argparse
import ast
import gzip
import hashlib
import importlib.util
import json
from pathlib import Path
import subprocess
import tempfile


def sha(path):
    digest = hashlib.sha256()
    with Path(path).open("rb") as source:
        for block in iter(lambda: source.read(1 << 20), b""):
            digest.update(block)
    return digest.hexdigest()


def load_module(path):
    spec = importlib.util.spec_from_file_location("native_profile_runner", path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def function_sources(path):
    source = Path(path).read_text()
    tree = ast.parse(source, filename=str(path))
    found = {}
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            found.setdefault(node.name, []).append(ast.get_source_segment(source, node))
    return found


def varint(value):
    if value < 0:
        value += 1 << 64
    output = bytearray()
    while value > 0x7f:
        output.append((value & 0x7f) | 0x80)
        value >>= 7
    output.append(value)
    return bytes(output)


def field(number, wire, value):
    tag = varint((number << 3) | wire)
    if wire == 0:
        return tag + varint(value)
    if wire == 2:
        return tag + varint(len(value)) + value
    raise AssertionError(wire)


def value_type(type_index, unit_index):
    return field(1, 0, type_index) + field(2, 0, unit_index)


def sample(values):
    return field(2, 2, b"".join(varint(value) for value in values))


def synthetic_profile(path):
    strings = ("", "cpu", "nanoseconds", "alloc_space", "bytes", "alloc_objects", "count")
    payload = b"".join((
        field(1, 2, value_type(1, 2)),
        field(1, 2, value_type(3, 4)),
        field(1, 2, value_type(5, 6)),
        field(2, 2, sample((20, 300, 4))),
        field(2, 2, sample((30, 700, 6))),
        *(field(6, 2, value.encode()) for value in strings),
    ))
    path.write_bytes(gzip.compress(payload))


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    root = Path(__file__).resolve().parent
    parser.add_argument("--runner", default=str(root / "qualify-proxy-native-go-profile.py"))
    parser.add_argument("--baseline-runner", required=True)
    parser.add_argument("--baseline-manifest", required=True)
    parser.add_argument("--lineage-manifest", default=str(root / "profile-derivative-manifest.json"))
    parser.add_argument("--output-parent", default=str(root))
    args = parser.parse_args()
    runner_path = Path(args.runner).resolve(strict=True)
    baseline_runner = Path(args.baseline_runner).resolve(strict=True)
    baseline_manifest_path = Path(args.baseline_manifest).resolve(strict=True)
    lineage_path = Path(args.lineage_manifest).resolve(strict=True)
    module = load_module(runner_path)
    lineage = json.loads(lineage_path.read_text())
    baseline_manifest = json.loads(baseline_manifest_path.read_text())
    output_parent = Path(args.output_parent).resolve(strict=True)
    work = Path(tempfile.mkdtemp(prefix="native-profile-contracts.", dir=output_parent))
    results = []

    def check(name, callback):
        callback()
        results.append({"name": name, "passed": True})

    def lineage_contract():
        assert lineage["kind"] == "p2pstream-native-go-profile-derivative-v1"
        assert lineage["classification"] == "diagnostic-only"
        assert lineage["baseline"]["source_ref"] == module.EXPECTED_BASE
        assert lineage["baseline"]["manifest_sha256"] == sha(baseline_manifest_path)
        assert lineage["baseline"]["runner_sha256"] == sha(baseline_runner)
        checkpoint = baseline_manifest_path.parent / "source-checkpoint.json"
        assert lineage["baseline"]["source_checkpoint_sha256"] == sha(checkpoint)
        assert lineage["baseline"]["binary_sha256"] == baseline_manifest["binary"]["sha256"]
        assert lineage["baseline"]["source_overlay_sha256"] == baseline_manifest["source_overlay"]["sha256"]
        assert lineage["derivative"]["runner_sha256"] == sha(runner_path)
        patch_path = lineage_path.parent / lineage["derivative"]["patch_path"]
        assert lineage["derivative"]["patch_sha256"] == sha(patch_path)
        reconstructed = work / "reconstructed-runner.py"
        completed = subprocess.run(["patch", "--batch", "--silent", "-o", reconstructed, baseline_runner, patch_path],
                                   stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=10)
        assert completed.returncode == 0, completed.stderr.decode(errors="replace")
        assert sha(reconstructed) == sha(runner_path)

    def exact_pair_contract():
        rows = [dict(role="production", public_scheme="http", concurrency=16),
                dict(role="native", public_scheme="http", concurrency=16)]
        module.validate_profile_row_chronology(rows)
        for invalid in (rows[:1], list(reversed(rows)), rows + [rows[0]],
                        [dict(role="production", public_scheme="https", concurrency=16), rows[1]],
                        [rows[0], dict(role="native", public_scheme="http", concurrency=128)]):
            try:
                module.validate_profile_row_chronology(invalid)
            except ValueError:
                continue
            raise AssertionError(f"invalid chronology accepted: {invalid!r}")

    def profile_flags_contract():
        for role in ("production", "native"):
            label = f"profile-{role}-http-h1-c16"
            assert module.profile_test_flags(label) == [
                f"-test.cpuprofile=/qualification/{label}.cpu.pprof",
                f"-test.memprofile=/qualification/{label}.mem.pprof"]
        for invalid in ("", "../escape", "/absolute", "Upper", "space value"):
            try:
                module.profile_test_flags(invalid)
            except ValueError:
                continue
            raise AssertionError(f"unsafe profile label accepted: {invalid!r}")

    def normalization_contract():
        assert module.warmup_timed_completions({"warmup_requests": 7, "useful_completed": 11}) == 18
        for invalid in ({}, {"warmup_requests": 0, "useful_completed": 1},
                        {"warmup_requests": 1, "useful_completed": 0},
                        {"warmup_requests": 1.5, "useful_completed": 2},
                        {"warmup_requests": True, "useful_completed": 2}):
            try:
                module.warmup_timed_completions(invalid)
            except ValueError:
                continue
            raise AssertionError(f"invalid normalization counts accepted: {invalid!r}")

    def profile_fail_closed_contract():
        valid = work / "valid.pprof"
        synthetic_profile(valid)
        assert module.profile_sample_totals(valid) == {
            "cpu:nanoseconds": 50,
            "alloc_space:bytes": 1000,
            "alloc_objects:count": 10,
        }
        empty = work / "empty.pprof"
        empty.write_bytes(b"")
        truncated = work / "truncated.pprof"
        truncated.write_bytes(valid.read_bytes()[:-2])
        inconsistent = work / "inconsistent.pprof"
        payload = field(1, 2, value_type(1, 2)) + field(2, 2, sample((1, 2)))
        payload += field(6, 2, b"") + field(6, 2, b"cpu") + field(6, 2, b"nanoseconds")
        inconsistent.write_bytes(gzip.compress(payload))
        for invalid in (work / "missing.pprof", empty, truncated, inconsistent):
            try:
                module.profile_sample_totals(invalid)
            except ValueError:
                continue
            raise AssertionError(f"partial profile accepted: {invalid}")

    def inherited_lifecycle_contract():
        baseline = function_sources(baseline_runner)
        derivative = function_sources(runner_path)
        exact_functions = (
            "resolve_command", "clean_qualification_environment", "qualification_child_environment",
            "attribution_resource_envelope", "attribution_load_environment", "parse_container_inventory",
            "validate_owned_container_absent", "validate_attribution_result_configuration",
            "process_group_alive", "terminate_process_group", "launch", "run_owned", "docker_command",
            "container_inventory", "require_container_absent", "inspect_container_identity", "await_ready",
            "await_socket", "stop_proxy", "stop_origin", "run_load")
        for name in exact_functions:
            assert baseline[name] == derivative[name], f"inherited lifecycle function changed: {name}"
        source = runner_path.read_text()
        for exact in (
            '"--memory", "4g", "--memory-swap", "4g"',
            '"--cpuset-cpus", proxy_cpus, "--pids-limit", "8192"',
            'proxy_cpus = "0,1,2,3"',
            'load_cpus = ",".join(map(str, range(4, 16)))',
            'P2PSTREAM_QUALIFICATION_STDLIB_MAX_IDLE="128"',
            'P2PSTREAM_QUALIFICATION_STDLIB_SHARDS="1"',
            'P2PSTREAM_QUALIFICATION_STDLIB_POOL="1"',
            'response_bytes") != 256 * result.get("attempted", 0)',
            'persisted_event_delta") != result["attempted"]',
            'process.get("cgroup_memory_max_bytes") != MEMORY_BYTES'):
            assert exact in source, f"missing inherited resource/correctness contract: {exact}"
        assert "P2PSTREAM_QUALIFICATION_START_PROFILES" not in source

    check("lineage-and-reconstruction", lineage_contract)
    check("exact-production-native-chronology", exact_pair_contract)
    check("symmetric-test-profile-flags", profile_flags_contract)
    check("warmup-plus-timed-normalization", normalization_contract)
    check("missing-partial-profile-fail-closed", profile_fail_closed_contract)
    check("inherited-lifecycle-and-resource-contracts-byte-identical", inherited_lifecycle_contract)
    output = work / "results.json"
    output.write_text(json.dumps({"passed": True, "contracts": results}, indent=2) + "\n")
    print(output)


if __name__ == "__main__":
    main()
