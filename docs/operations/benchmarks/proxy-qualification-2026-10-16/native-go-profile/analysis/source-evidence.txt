    21	// Derive the fixed multirow statement from the generated single-row SQL so
    22	// the column contract remains in sql/query.sql. Only the fixed VALUES tuple
    23	// is repeated; request data is always bound through statement parameters.
    24	var insertProxyRequestEventsBatch = func() string {
    25		prefix, values, ok := strings.Cut(insertProxyRequestEventAt, " VALUES ")
    26		if !ok {
    27			panic("generated proxy request event insert is missing VALUES")
    28		}
    29		tuple := strings.TrimSpace(values)
    30		return prefix + " VALUES " + strings.Repeat(tuple+",", proxyRequestEventInsertBatchRows-1) + tuple
    31	}()

    33	func PrepareProxyRequestEventInserter(ctx context.Context, tx *sql.Tx) (*ProxyRequestEventInserter, error) {
    34		if tx == nil {

    38		if err != nil {
    39			return nil, err

    41		return &ProxyRequestEventInserter{tx: tx, stmt: stmt}, nil

    44	func (i *ProxyRequestEventInserter) Insert(ctx context.Context, arg InsertProxyRequestEventAtParams) error {
    45		if i == nil || i.stmt == nil {
    46			return errors.New("proxy request event inserter is not prepared")
    47		}
    48		var args [23]any
    49		_, err := i.stmt.ExecContext(ctx, appendProxyRequestEventArgs(args[:0], arg)...)

    53	// InsertBatch amortizes SQLite execution and cancellation coordination over
    54	// bounded groups. All groups and the short final remainder use the caller's
    55	// existing transaction and cancellation context.
    56	func (i *ProxyRequestEventInserter) InsertBatch(ctx context.Context, events []InsertProxyRequestEventAtParams) error {
    57		if i == nil || i.stmt == nil {
    58			return errors.New("proxy request event inserter is not prepared")
    59		}
    60		if len(events) >= proxyRequestEventInsertBatchRows && i.batchStmt == nil {
    61			stmt, err := i.tx.PrepareContext(ctx, insertProxyRequestEventsBatch)
    62			if err != nil {
    63				return err
    64			}
    65			i.batchStmt = stmt
    66		}
    67		if len(events) >= proxyRequestEventInsertBatchRows {
    68			var storage [proxyRequestEventInsertBatchRows * 23]any
    69			for len(events) >= proxyRequestEventInsertBatchRows {
    70				args := storage[:0]
    71				for _, event := range events[:proxyRequestEventInsertBatchRows] {
    72					args = appendProxyRequestEventArgs(args, event)
    73				}
    74				if _, err := i.batchStmt.ExecContext(ctx, args...); err != nil {
    75					return err
    76				}
    77				events = events[proxyRequestEventInsertBatchRows:]
    78			}
    79		}
    80		for _, event := range events {
    81			if err := i.Insert(ctx, event); err != nil {
    82				return err
    83			}
    84		}
    85		return nil
    86	}

   410	func (r *observabilityRecorder) run() {
   411		ticker := time.NewTicker(observabilityRecorderFlushInterval)
   412		defer ticker.Stop()
   413		batch := make([]db.InsertProxyRequestEventAtParams, 0, observabilityRecorderMaxBatch)
   414		touches := make(map[publicCacheTouchKey]*publicCacheTouchState)
   415		for {
   416			events := r.events
   417			if len(batch) >= observabilityRecorderMaxBatch {
   418				// A failed batch must stay fixed-size while it is retried. Leaving
   419				// additional events in the bounded channel prevents an outage from
   420				// turning the batch into unbounded heap growth.
   421				events = nil
   422			}
   423			select {
   424			case event := <-events:
   425				batch = append(batch, event)
   426				r.pendingEvents.Store(int64(len(batch)))
   427				if len(batch) >= observabilityRecorderMaxBatch {
   428					_ = r.runBackgroundFlush(func(ctx context.Context) error {
   429						return r.flushBatch(ctx, &batch, touches, time.Now().UTC(), false)
   430					})
   431				}
   432			case touch := <-r.touches:
   433				r.queuePublicCacheTouch(touches, touch)
   434			case req := <-r.control:
   435				eventCount := len(r.events)
   436				touchCount := len(r.touches)
   437				err := r.runFlushOperation(req.ctx, req.stop, func(ctx context.Context) error {
   438					return r.flushQueued(ctx, &batch, touches, eventCount, touchCount)
   439				})
   440				if !req.stop {
   441					req.done <- err
   442					continue
   443				}
   444				if err != nil {
   445					r.mu.Lock()
   446					r.restoreCloseAttemptLocked(req.closeAttempt)
   447					r.mu.Unlock()
   448					req.done <- err
   449					continue
   450				}
   451				r.pendingEvents.Store(0)
   452				r.pendingTouches.Store(0)
   453				r.mu.Lock()
   454				if r.closeAttempt == req.closeAttempt {
   455					r.closeAttempt = nil
   456					close(r.done)
   457					close(req.closeAttempt)
   458				}
   459				r.mu.Unlock()
   460				req.done <- nil
   461				return
   462			case <-ticker.C:
   463				now := time.Now().UTC()
   464				r.prunePublicCacheTouches(touches, now)
   465				_ = r.runBackgroundFlush(func(ctx context.Context) error {
   466					return r.flushBatch(ctx, &batch, touches, now, false)
   467				})
   468			}
   469		}

   472	func (r *observabilityRecorder) flushQueued(
   473		ctx context.Context,

   476		eventCount int,

   496		for i := 0; i < touchCount; i++ {
   497			r.queuePublicCacheTouch(touches, <-r.touches)
   498		}
   499		return r.flushBatch(ctx, batch, touches, time.Now().UTC(), true)

   502	func (r *observabilityRecorder) queuePublicCacheTouch(touches map[publicCacheTouchKey]*publicCacheTouchState, queued publicCacheTouch) {
   503		if queued.key.keyDigest == "" || queued.key.storedAt.IsZero() || queued.accessedAt.IsZero() {
   504			return
   505		}
   506		queued = newPublicCacheTouch(queued.key.keyDigest, queued.key.storedAt, queued.accessedAt)
   507		if touch := touches[queued.key]; touch != nil {
   508			if touch.hitCount == math.MaxInt64 {
   509				r.dropTouch()
   510				return
   511			}
   512			touch.hitCount++
   513			if queued.accessedAt.After(touch.lastAccessedAt) {
   514				touch.lastAccessedAt = queued.accessedAt
   515			}
